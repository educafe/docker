#Press Enter key to all prompting questions

#Copy public key onto node servers
#Input 'yes' to prompting questions

ssh-copy-id -i ~/.ssh/id_rsa educafe@192.168.56.106

#Verify connection
ssh educafe@192.168.56.105
ssh educafe@192.168.56.106
